<script>
  const canvas = document.getElementById('matrixCanvas');
  const ctx = canvas.getContext('2d');

  // Set sharp internal resolution
  function resizeCanvas() {
    canvas.width = canvas.clientWidth;
    canvas.height = canvas.clientHeight;
  }
  resizeCanvas();

  // Katakana, Latin, and Numeric characters
  const alphabet = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789ｱｲｳｴｵｶｷｸｹｺｻｼｽｾｿﾀﾁﾂﾃﾄﾅﾆﾇﾈﾉﾊﾋﾌﾍﾎﾏﾐﾑﾒﾓﾔﾕﾖﾗﾘﾙﾚﾛﾜﾝ';
  const fontSize = 16;
  const columns = Math.floor(canvas.width / fontSize);

  // Initialize falling positions (y-coordinates)
  const rainDrops = Array(columns).fill(1);

  function drawMatrix() {
    // Semi-transparent black background creates the trailing fade effect
    ctx.fillStyle = 'rgba(0, 0, 0, 0.05)';
    ctx.fillRect(0, 0, canvas.width, canvas.height);

    ctx.font = fontSize + 'px monospace';

    for (let i = 0; i < rainDrops.length; i++) {
      // Pick a random character
      const text = alphabet.charAt(Math.floor(Math.random() * alphabet.length));
      
      // X coordinate is column index * font size
      const x = i * fontSize;
      // Y coordinate is the drop position * font size
      const y = rainDrops[i] * fontSize;

      // Make the very tip of the drop bright white, and the trail green
      if (Math.random() > 0.98) {
        ctx.fillStyle = '#fff';
      } else {
        ctx.fillStyle = '#0F0';
      }

      ctx.fillText(text, x, y);

      // Reset drop to the top randomly after it leaves the screen
      if (y > canvas.height && Math.random() > 0.975) {
        rainDrops[i] = 0;
      }
      rainDrops[i]++;
    }
  }

  // Adjust columns if the user resizes their window
  window.addEventListener('resize', () => {
    resizeCanvas();
    while(rainDrops.length < Math.floor(canvas.width / fontSize)) {
      rainDrops.push(0);
    }
  });

  // Run the loop at ~30 FPS
  setInterval(drawMatrix, 33);
</script>
